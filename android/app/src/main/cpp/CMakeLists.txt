cmake_minimum_required(VERSION 3.24)
project(WiizVAndroid LANGUAGES C CXX)
get_filename_component(WIIZV_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(CORE_DIR "${WIIZV_ROOT}/third_party/snes9x/upstream")
set(LIBRETRO_COMM_DIR "${CORE_DIR}/libretro/libretro-common")
file(READ "${CORE_DIR}/libretro/Makefile.common" CORE_MAKEFILE)
string(REGEX MATCHALL "\\$\\((CORE_DIR|LIBRETRO_COMM_DIR)\\)/[A-Za-z0-9_/]+\\.(cpp|c)" CORE_FILES "${CORE_MAKEFILE}")
foreach(SOURCE IN LISTS CORE_FILES)
    string(REPLACE "$(CORE_DIR)" "${CORE_DIR}" SOURCE "${SOURCE}")
    string(REPLACE "$(LIBRETRO_COMM_DIR)" "${LIBRETRO_COMM_DIR}" SOURCE "${SOURCE}")
    list(APPEND CORE_SOURCES "${SOURCE}")
endforeach()
list(REMOVE_DUPLICATES CORE_SOURCES)
add_library(wiizv_snes9x SHARED ${CORE_SOURCES})
target_compile_features(wiizv_snes9x PRIVATE cxx_std_17)
target_compile_definitions(wiizv_snes9x PRIVATE ANDROID __LIBRETRO__ ALLOW_CPU_OVERCLOCK)
target_compile_options(wiizv_snes9x PRIVATE -O2 -include "${WIIZV_ROOT}/third_party/snes9x/wiizv_linkage.h")
target_include_directories(wiizv_snes9x PRIVATE "${CORE_DIR}" "${CORE_DIR}/libretro" "${LIBRETRO_COMM_DIR}/include" "${CORE_DIR}/apu" "${CORE_DIR}/apu/bapu")
target_link_libraries(wiizv_snes9x PRIVATE log m)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory("${WIIZV_ROOT}/third_party/SDL3/SDL3-3.2.28" "${CMAKE_CURRENT_BINARY_DIR}/sdl")
add_library(main SHARED "${WIIZV_ROOT}/app/main.cpp" "${CORE_DIR}/sha256.cpp")
target_compile_features(main PRIVATE cxx_std_20)
target_compile_options(main PRIVATE -O2)
target_include_directories(main PRIVATE "${WIIZV_ROOT}/third_party/snes9x")
target_link_libraries(main PRIVATE SDL3::SDL3 log)
